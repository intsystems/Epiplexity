\begin{frame}{Epiplexity: information accessible to a bounded learner}
Epiplexity is the structural information in data that an observer limited to time $T$ can extract \citep{finzi2026}.

\medskip
Minimise a two-part code over models $P\in\mathcal P_T$ that run in time $T$:
\[
P^* = \arg\min_{P\in\mathcal P_T}\bigl\{|P| + \E[-\log P(X)]\bigr\}.
\]
The minimum splits into structure and randomness:
\[
\underbrace{\MDL_T(X)}_{\text{total}}
=
\underbrace{|P^*|}_{S_T(X):\ \text{epiplexity}}
+
\underbrace{\E[-\log P^*(X)]}_{H_T(X):\ \text{time-bounded entropy}}
\]
\begin{itemize}
\item $S_T$ depends on the bound $T$ and on the class $\mathcal P_T$.
\item Neural training approximates the minimisation over $P$.
\end{itemize}
\end{frame}

\begin{frame}{What we test: the estimators, not the architectures}
The same estimators run in three setups (DiT: diffusion transformer):
\[
X \xrightarrow{\ \text{setup } a\ } P_T,
\qquad
a \in \{\text{MLP},\ \text{CNN},\ \text{DiT}+\text{sampler}\}.
\]
\medskip
Two sources of change have to be told apart:
\begin{itemize}
\item \emph{manipulations} inside a setup: data order, pixel permutation, sampler schedule, bound $T$;
\item the \emph{observer family}: hypothesis class and inductive bias.
\end{itemize}

\medskip
\textbf{Goal:} which part of each estimate comes from the data, which from the setup.
\end{frame}
% ------------------------------------------------------------
\begin{frame}{Estimators: one learning trajectory, several codes}
\small
Model $P_i$ is trained on $z_0,\dots,z_{i-1}$ out of $M$ examples; $\ell_i(z)=-\log P_i(z)$.

\medskip
\textbf{Prequential} \citep{finzi2026}: area of the loss curve above the final loss.
\[
S_{\mathrm{preq}} \approx \sum_{i<M}\bigl[\ell_i(z_i)-\ell_M(z_i)\bigr]
\]
\textbf{Requential} \citep{qiu2026}: student s trained on samples of teacher t.
\[
S_{\mathrm{req}} \approx \sum_i \KL\bigl(P_i^{\mathrm{t}}\,\big\|\,P_i^{\mathrm{s}}\bigr)
\]
\textbf{EDL} \citep{donoway2026}: floor is the population loss $L$ of the final $P_T$.
\[
\mathrm{EDL} = \E\sum_{i<M}\ell_i(z_i) - M\,L(P_T)
\]
\medskip
EDL fixes the training algorithm; epiplexity minimises over $\mathcal P_T$.
\end{frame}

\begin{frame}{Experimental design: three setups, one manipulation each}
Benchmark: elementary cellular automata (ECA), as in \citet{finzi2026}. The rule sets how much learnable structure the data holds.
\begin{center}
\small
\setlength{\tabcolsep}{4pt}
\renewcommand{\arraystretch}{1.12}
\begin{tabular}{lccc}
\toprule
& MLP & CNN & DiT \\
\midrule
Parameters & $10^3$--$10^6$ & $10^3$--$10^6$ & small $\rightarrow$ medium \\
Prior & weakest & locality & generative, iterative \\
Optimizer & SGD / Adam(W) & SGD / Adam(W) & Adam(W) \\
Manipulation & example order & pixel permutation & sampler, \#steps \\
\bottomrule
\end{tabular}
\end{center}
\begin{itemize}
\item MLP: no spatial prior; example order is part of the observer.
\item CNN: fixed pixel permutation; labels kept, locality removed.
\item DiT: the sampler is part of the observer.
\end{itemize}
Within one comparison the task and $T$ are fixed.
\end{frame}
% ------------------------------------------------------------
\begin{frame}{Validation}
For each task $d$, setup $a$, bound $T$ and seed $s$ we record the training run and compute $S_{\mathrm{preq}}$, $S_{\mathrm{req}}$ and EDL.
\begin{itemize}
\item \textbf{Agreement:} Spearman and Kendall $\tau$ between task orderings, within and across setups.
\item \textbf{Setup dependence:} $\Delta_{ab}(d,T) = S(d,a,T) - S(d,b,T)$.\
Large $\Delta$ with a stable task order: the estimate partly measures the setup.
\item \textbf{Seed variance}, compared with the spread across setups.
\item \textbf{Baselines:} final test loss, sample efficiency.
\end{itemize}

\medskip
\textbf{Deliverable:} a map of where each estimator is setup-invariant.
\end{frame}
