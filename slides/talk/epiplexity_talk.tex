% ============================================================ MOTIVATION
\begin{frame}{Motivation: evaluating pretraining data}
\begin{columns}[T]
\begin{column}{0.50\textwidth}
\small
Pretraining is intended to improve performance on tasks beyond the training distribution.

\medskip
Dataset size and final predictive loss do not quantify how much reusable structure a model acquires.

\begin{block}{The paper's question}
How can we measure structural information in data relative to a fixed computational budget?
\end{block}
\end{column}
\begin{column}{0.46\textwidth}
\centering
\includegraphics[width=\linewidth,trim=0 0 660 40,clip]{figs/fig1.pdf}
\raggedright
\src{Schematic loss curves. Dashed line: final loss. Green area: cumulative excess loss above that baseline, used as a structural-information proxy. Fig.~1, \citet{finzi2026}.}
\end{column}
\end{columns}
\end{frame}
% ------------------------------------------------------------
\begin{frame}{Shannon entropy and Kolmogorov complexity}
\small
\begin{columns}[T]
\begin{column}{0.47\textwidth}
\textbf{Shannon entropy: a distribution}
\[
\begin{aligned}
p(x)&=\Pr(X=x),\\[-0.15em]
H(X)&=\E[-\log_2 p(X)].
\end{aligned}
\]
For discrete $X$, entropy is a lower bound on the expected length of any binary prefix code.

\medskip
Uniform $n$-bit strings: $H(X)=n$.

\smallskip
A fixed string: $H(X)=0$.
\end{column}
\begin{column}{0.47\textwidth}
\textbf{Kolmogorov complexity: a string}
\[
K(x)=\min_{p:\,\mathcal U(p)=x}|p|.
\]
The shortest self-delimiting program that outputs $x$ and halts on a fixed prefix-free universal Turing machine $\mathcal U$.

\medskip
An alternating string is specified by its rule and length. Most uniform $n$-bit strings require at least $n-O(1)$ program bits.
\end{column}
\end{columns}
\medskip
\src{$K(x)$ is uncomputable in general. Definition~1 and Section~2.1, \citet{finzi2026}.}
\end{frame}
% ============================================================ FORMAL PROBLEM STATEMENT
\begin{frame}{Why introduce a computation bound?}
\small
\begin{block}{A short program can require a long computation}
A simulator and its initial conditions can specify a complex trajectory. Reproducing it may still require more computation than the observer can perform.
\end{block}

\medskip
Neither Shannon entropy nor unbounded Kolmogorov complexity restricts the runtime of a predictive model.

\medskip
The paper therefore restricts the admissible models to programs that evaluate probabilities and generate samples within time $T$.

\medskip
Within this class, it separates the description of the model from the description of data given the model.
\end{frame}
% ------------------------------------------------------------
\begin{frame}{Time-bounded probabilistic models}
\small
Fix a prefix-free universal Turing machine $\mathcal U$ and a time bound $T(n)$.
A program $\prog\in\PT$ implements a distribution $P$ on $\{0,1\}^n$ with two modes:
\begin{itemize}\setlength{\itemsep}{0.7em}
\item \textbf{Evaluation:} given $x$, return $P(x)$ within $T(n)$ steps.
\item \textbf{Sampling:} given independent random bits, generate $x\sim P$ within $T(n)$ steps.
\end{itemize}

\medskip
The returned probabilities must sum to one and agree with the sampling distribution.
Roman $\prog$ denotes the program; italic $P$ denotes its distribution. $|\prog|$ is the program length in bits.

\medskip
\src{Definition~7, \citet{finzi2026}. The formal bound is non-decreasing and time-constructible.}
\end{frame}
% ------------------------------------------------------------
\begin{frame}{Epiplexity: the two-part description}
\small
For $X$ on $\{0,1\}^n$, choose an admissible model minimizing the expected two-part description length:
\[
\prog^\star\in\argmin_{\prog\in\PT}
\bigl\{|\prog|+\E[-\log_2 P(X)]\bigr\}.
\]
Among minimizers, select one with the shortest program.

\begin{columns}[T]
\begin{column}{0.47\textwidth}
\begin{block}{Epiplexity: model description}
\[
S_T(X):=|\prog^\star|.
\]
Length of the selected model's program.
\end{block}
\end{column}
\begin{column}{0.47\textwidth}
\begin{block}{Time-bounded entropy: data description}
\[
H_T(X):=\E[-\log_2 P^\star(X)].
\]
Expected data-code length using that model.
\end{block}
\end{column}
\end{columns}
\[
\MDL_T(X)=S_T(X)+H_T(X).
\]
\src{Definition~8, \citet{finzi2026}.}
\end{frame}
% ------------------------------------------------------------
\begin{frame}{Examples: random bits and a repeated pattern}
\small
Assume the length $n$ is specified and the time bound permits the models below.

\begin{block}{Uniform bits: $X\sim U_n$}
A constant-length program assigns $P(x)=2^{-n}$ to every $n$-bit string.
\[
\MDL_T(X)\le n+c,\qquad H_T(X)\ge H(X)=n.
\]
Consequently $S_T(X)\le c$: a large data-code length can accompany a small model description.
\end{block}

\begin{block}{Alternating strings: two equally likely outcomes}
$X=0101\ldots$ or $1010\ldots$, each with probability $\tfrac12$.
A constant-length program generates either pattern in linear time.
\[
S_T(X)=O(1),\qquad H_T(X)=O(1).
\]
Here both the model description and the residual data code are small.
\end{block}
\end{frame}
% ============================================================ THEORY
\begin{frame}{Cryptographic consequences of the time bound}
\small
\begin{itemize}\setlength{\itemsep}{0.65em}
\item \textbf{Pseudorandom generator.} $G$ maps $k$ random seed bits to $n>k$ output bits.
      $H(G(U_k))\le k$, yet for a secure generator $H_{\Poly}(G(U_k))\approx n$ and $S_{\Poly}(G(U_k))=O(1)$.
\item \textbf{One-way function.} Computing $f(x)$ is efficient; finding a preimage of $f(x)$ is hard for efficient algorithms.
      For a one-way \emph{permutation}, the paper proves that the two factorization orders have different time-bounded entropies.
\end{itemize}

\medskip
\src{Theorems~9 and~13, \citet{finzi2026}. These results assume security against non-uniform probabilistic polynomial-time algorithms.}
\end{frame}
% ------------------------------------------------------------
\begin{frame}{Conditional epiplexity: predicting labels from inputs}
\small
For supervised data $(X,Y)$, the decoder already has the input $X$.
The model evaluates $P(y\mid x)$ and samples $Y\mid X=x$ within the bound for each fixed $x$.

\[
\prog^\star_{Y\mid X}\in\argmin_{\prog\in\mathcal P_T^X}
\left\{|\prog|+\E_{X,Y}[-\log_2 P(Y\mid X)]\right\}.
\]
\[
S_T(Y\mid X):=|\prog^\star_{Y\mid X}|,\qquad
H_T(Y\mid X):=\E_{X,Y}[-\log_2 P^\star(Y\mid X)].
\]
\begin{block}{Example: image classification}
The model code describes the classifier. The data code describes the labels given the images; it does not encode the images.
\end{block}
These quantities are defined directly. In general,
$H_T(Y\mid X)\ne H_T(X,Y)-H_T(X)$.

\medskip
\src{Definition~11, \citet{finzi2026}.}
\end{frame}
% ============================================================ METHOD
\begin{frame}{From programs to neural-network estimates}
\begin{columns}[T]
\begin{column}{0.58\textwidth}
\small
\begin{itemize}\setlength{\itemsep}{0.25em}
\item Search over all programs is intractable, so fix an architecture and optimiser.
\item Approximate dense-model accounting: $6ND+2N\Dt\le T$, where $N$ counts parameters and $D,\Dt$ count training and evaluation tokens.
\item Each run gives a model-part estimate plus a validation-data code. Select the shortest admissible two-part code; a lower hull smooths the empirical curve.
\item Report its model and data parts as estimates $\widehat S_T$ and $\widehat H_T$.
\end{itemize}
\end{column}

\begin{frame}{Estimating the model part: prequential coding}
\begin{columns}[T]
\begin{column}{0.64\textwidth}
\small
\begin{itemize}\setlength{\itemsep}{0.2em}
\item $P_i$ has seen $Z_0,\ldots,Z_{i-1}$; $P_0$ is the initial model.
\item Encode $Z_i$ using its predicted probability, then update. The decoder repeats the update.
\item The sequential data code sums these costs. Subtract the final-model data code to estimate the model part:
\[
\widehat K_{\mathrm{preq}}=\sum_{i<M}\bigl[-\log_2 P_i(Z_i)+\log_2 P_M(Z_i)\bigr].
\]
\end{itemize}
\end{column}
\begin{column}{0.32\textwidth}
\vspace{0.5em}
\centering
\includegraphics[width=\linewidth]{figs/requential_illustration.pdf}
\raggedright
\src{Blue area: prequential model-part estimate. Red area: teacher--student loss gap used to visualise requential coding. Fig.~2a, \citet{finzi2026}.}
\end{column}
\end{columns}
\small
\begin{itemize}\setlength{\itemsep}{0.2em}
\item On an i.i.d.\ stream, the pre-update loss estimates loss on a fresh sample.
\item The area is a heuristic model-part estimate; subtraction alone gives no model-only code.
\end{itemize}
\end{frame}
% ------------------------------------------------------------
\begin{frame}{An explicit model code: requential coding}
\small
\begin{itemize}\setlength{\itemsep}{0.25em}
\item A teacher model trains on real data; a student model trains on teacher-distributed samples \citep{qiu2026}.
\item Both sides share the student and a random seed. Send the index of a student proposal selected to have the teacher's distribution.
\item Relative entropy coding bounds the expected message length by a function of the teacher--student KL divergence:
\[
\E[\ell_i\mid\mathcal F_i]\le\KL_i+2\log_2(1+\KL_i)+\kappa,\quad \kappa<5.21,
\]
Here $\KL_i=\E_{Z\sim P_i^{\mathrm t}}[\log_2(P_i^{\mathrm t}(Z)/P_i^{\mathrm s}(Z))]$ is the Kullback--Leibler divergence, and $\mathcal F_i$ is the prior history. $\sum_i\KL_i$ is the leading model-code estimate.
\end{itemize}
\centering
\begin{tabular}{@{}lll@{}}
\toprule
& Prequential & Requential \\
\midrule
Model part & heuristic subtraction & explicit student reconstruction \\
Training & one network & teacher and student, $2$--$10\times$ slower \\
Reported values & larger in the paper & smaller in the paper \\
\bottomrule
\end{tabular}

\end{frame}

% ============================================================ SUMMARY
\begin{frame}{Conclusions and limitations}
\textbf{Summary}
\begin{itemize}\setlength{\itemsep}{0.3em}
\item $S_T$ counts selected model bits; $H_T$ is the expected data-code length.
\item Computational restrictions can change both the total description length and its decomposition.
\item Neural-network estimates combine a model-code estimate with validation loss, optimized under a compute budget.
\end{itemize}
\medskip
\textbf{Limitations}
\begin{itemize}\setlength{\itemsep}{0.3em}
\item $S_T$ is not a guarantee of downstream usefulness.
\item The existence theorem is conditional on cryptographic assumptions and proves $\Omega(\log_2 n)$ growth.
\item Estimates depend on architecture, optimiser, $T$, dataset size and random seed.
\end{itemize}
\end{frame}

% ============================================================ QUESTIONS
\begin{frame}{Questions for the audience: answers}
\small
\begin{enumerate}\setlength{\itemsep}{1.1em}
\item Let $X\sim U_n$. Which quantity is close to $n$: $S_T(X)$ or $H_T(X)$?
\begin{block}{Answer}
$H_T(X)=n+O(1)$. A constant-length uniform model gives $\MDL_T(X)\le n+c$; since $H_T(X)\ge n$, the selected model has $S_T(X)\le c$.
\end{block}
\item In the prequential estimate, which part of the training-loss curve approximates the epiplexity?
\begin{block}{Answer}
The cumulative excess loss above the final-model baseline:
$\sum_i[-\log_2 P_i(Z_i)+\log_2 P_M(Z_i)]$.
\end{block}
\item What does changing $T$ change?
\begin{block}{Answer}
It enlarges the admissible model class. The minimum total $\MDL_T$ cannot increase, while the selected model length $S_T$ may rise or fall.
\end{block}
\end{enumerate}
\end{frame}
% ============================================================ BACKUP
\begin{frame}{Backup: basic properties}
\begin{proposition}[Sec.~3 of the paper]
Let $c_1,c_2$ be constants independent of $X$. For a random variable $X$ on $\{0,1\}^n$:
\begin{enumerate}
\item $S_T(X)\ge0$ and $H_T(X)\ge0$;
\item $H(X)\le\MDL_T(X)\le n+c_1$;
\item $\MDL_{T'}(X)\le\MDL_T(X)$ whenever $T'(n)\ge T(n)$;
\item $\MDL_{T'}(f^{-1}(X))\le\MDL_T(X)+|\mathrm f|+c_2$ with $T'(n)=T(n)+\mathrm{Time}(\mathrm f)$, for a program $\mathrm f$ that computes a bijection $f$ in fixed time.
\end{enumerate}
\end{proposition}
Property~4 is the analogue of $K(f(x))\le K(x)+K(f)+c$. Under a time bound a short program for $f^{-1}$ gives no short program for $f$.
\begin{theorem}[Thm.~10 of the paper]
If one-way functions secure against probabilistic polynomial-time algorithms with polynomial advice exist, then some random variables $X_n$ on $\{0,1\}^n$ have $S_{\Poly}(X_n)=\Omega(\log_2 n)$.
\end{theorem}
\end{frame}

\begin{frame}{Backup: epiplexity need not increase with compute}
\begin{columns}[T]
\begin{column}{0.52\textwidth}
\small
\textbf{Setting.} ECA rule~54, $64$ updates. The non-looped model predicts the final state; the looped model simulates each update.

\medskip
\begin{itemize}\setlength{\itemsep}{0.2em}
\item Below $10^{15}$ FLOPs, the non-looped model gives the shorter code; its model part reaches $1.45\times10^7$ bits.
\item Above that bound, the looped model wins because the rule is a compact model.
\item Thus epiplexity need not increase with the budget; growth is expected to be more typical for natural data.
\end{itemize}
\end{column}

\centering

\take{With less compute the best model is a longer program: $S_T$ need not grow with $T$.}
