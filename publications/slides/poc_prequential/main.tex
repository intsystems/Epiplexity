\begin{frame}{Research question}
\begin{block}{Question}
Is the amount of structure learned from a synthetic generator associated with forecast accuracy on real time series?
\end{block}\vspace{2mm}
\begin{columns}[T]
\column{.49\textwidth}\textbf{For each generator \(g\)}
\begin{itemize}\item Train Moirai from scratch.\item Compute the estimate \(\widehat S_g\).\item Evaluate transfer on GIFT-Eval.\end{itemize}
\column{.47\textwidth}\textbf{Comparison protocol}
\begin{itemize}\item One architecture.\item Equal budgets within a series.\item 11 generators and 55 tasks.\end{itemize}
\end{columns}\vspace{2mm}
\par\vspace{1mm}{\small \(g\longmapsto(\widehat S_g,Q_g),\quad\rho=\operatorname{Spearman}(\widehat S_g,Q_g).\)}\par\vspace{1mm}
\smallnote{\(Q_g\) is forecast error, where lower is better. We expect a useful association to be negative.}
\end{frame}

\begin{frame}[fragile]{Prequential coding: predict, then train}
For window \(i\), let \(C_i\) be the context, \(Y_i\) the target, and \(b(i)\) the batch index.
\[\ell_i=-\log q_{\theta_{b(i)-1}}(Y_i\mid C_i),\qquad L_{\mathrm{pre}}=\sum_{i=1}^{N}\ell_i.\]
\begin{lstlisting}
for batch in stream:
    per_window, _ = self.compute_loss(batch)
    loss = per_window.mean() / self.spec.horizon
    self.optimizer.zero_grad(set_to_none=True)
    loss.backward()
    self.optimizer.step()
    # persist per_window: pre-update values
\end{lstlisting}
\smallnote{In blockwise prequential coding, the same model scores every window in a batch. Storing the losses after the update preserves their pre-update values. The example omits clipping and EMA.}
\vspace{1mm}{\small \textbf{Key condition.} Score each window before training on its batch.}
\end{frame}

\begin{frame}{From prequential loss to epiplexity}
Prequential loss includes errors made while the model is still learning. Subtract the final held-out loss scaled to the same number of windows.
\vspace{-1mm}
\[\overline\ell_{\mathrm{fin}}=\frac1m\sum_{j=1}^{m}-\log q_{\bar\theta_{\mathrm{fin}}}(Y_j^{\mathrm{held}}\mid C_j^{\mathrm{held}})\]
\begin{block}{Decomposition used by episynth}
\[\widehat H=\frac{N\overline\ell_{\mathrm{fin}}}{\ln2},\quad\widehat S=\frac{L_{\mathrm{pre}}-N\overline\ell_{\mathrm{fin}}}{\ln2},\quad\widehat S+\widehat H=\frac{L_{\mathrm{pre}}}{\ln2}.\]
\end{block}
\smallnote{\(\bar\theta_{\mathrm{fin}}\) denotes the final EMA weights and \(m\) counts held-out windows independent of training. Losses are in nats and the result is in bits.}
\vspace{1mm}{\small\textbf{An empirical estimate of theoretical epiplexity.}}
\end{frame}

\begin{frame}[fragile]{One protocol for integrating estimators}

class EpiplexityEstimator(Protocol):
    name: str

\item Register the class with \code{@register("name")}.

\end{enumerate}\vspace{1mm}

\begin{frame}{Appendix: uncertainty and limitations}
\[\operatorname{SE}(\widehat S)=\frac{N}{\ln2}\operatorname{SE}(\overline\ell_{\mathrm{fin}}),\qquad\operatorname{SE}(\overline\ell_{\mathrm{fin}})=\frac{s_{\mathrm{held}}}{\sqrt m}.\]
\begin{itemize}
\item With independent held-out windows, this error reflects the finite held-out sample. Variation across training seeds is separate.
\item EMA and raw endpoints can yield different areas. Fix the protocol and check sensitivity.
\item With continuous densities, NLL and its components can be negative. Scale and dequantization matter.
\item An estimate from one run depends on its architecture and budget. The theoretical definition optimizes over observers.
\end{itemize}
\smallnote{A negative signed area alone offers insufficient evidence for negative theoretical epiplexity.}
\end{frame}
